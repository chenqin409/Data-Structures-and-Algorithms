typedef struct Bnode
{
    char data;
    struct Bnode *lchild,*rchild;
}Bnode,*Btree;

void createtree(Btree &T)
{
    char ch;
    scanf("%c",&ch);
    if (ch=='#')
        T=NULL;
    else{
        T=new Bnode;
        T->data=ch;
        createtree(T->lchild);
        createtree(T->rchild);

    }
}

int countnode(Btree T)
{
   if (T==NULL)
        return 0;
    else return countnode(T->lchild)+countnode(T->rchild)+1;//加根结点
}
int countleaf(Btree T)
{   if (T==NULL)
        return 0;
    else{
        if ((!T->lchild)&&(!T->rchild)) return 1;
        else return countleaf(T->lchild)+countleaf(T->rchild);

    }

}
Btree pre_mid_create(char *pre,char *mid,int len)
{
    if (len==0)
         return NULL;
    char ch=pre[0];
    int index=0;
    while(mid[index]!=ch)
        {index++;}
    Btree T=new Bnode;
    T->data=ch;
    T->lchild=pre_mid_create(pre+1,mid,index);
    T->rchild=pre_mid_create(pre+index+1,mid+index+1,len-index-1);
   return T;
}

Btree post_mid_create(char *last,char *mid,int len)
{
   if (len==0)
         return NULL;
    char ch=last[len-1];
    int index=0;
    while(mid[index]!=ch)
        {index++;}
    Btree T=new Bnode;
    T->data=ch;
     T->lchild=post_mid_create(last,mid,index);
     T->rchild=post_mid_create(last+index,mid+index+1,len-index-1);
      return T;
}
